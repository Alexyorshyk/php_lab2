<?php
$uploadDir = __DIR__ . DIRECTORY_SEPARATOR . 'uploads';
$files = [];
if (is_dir($uploadDir)) {
    foreach (scandir($uploadDir) as $file) {
        if ($file !== '.' && $file !== '..' && is_file($uploadDir . DIRECTORY_SEPARATOR . $file)) $files[] = $file;
    }
    sort($files, SORT_NATURAL | SORT_FLAG_CASE);
}
?>
<!DOCTYPE html><html lang="uk"><head><meta charset="UTF-8"><title>Список файлів</title></head><body>
<h1>Файли у директорії uploads</h1>
<?php if (!$files): ?><p>Файлів ще немає.</p><?php else: ?><ul>
<?php foreach ($files as $file): ?><li><?= htmlspecialchars($file, ENT_QUOTES, 'UTF-8') ?>
(<a href="uploads/<?= rawurlencode($file) ?>" download>завантажити</a>)</li><?php endforeach; ?>
</ul><?php endif; ?><p><a href="index.html">На головну</a></p></body></html>
